### Phase 2: Geometry-Aware Graph Builder (Claim 2 Compliance)

Build 3D attributed graphs $G = (V, E, X, R)$ using RDKit's ETKDG conformer generator:

```python
def mol_to_spatial_graph(smiles: str) -> Data:
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    mol = Chem.AddHs(mol)
    res = AllChem.EmbedMolecule(mol, AllChem.ETKDGv3())
    if res != 0: # Conformer generation fallback
        AllChem.Compute2DCoords(mol)
        pos = torch.zeros((mol.GetNumAtoms(), 3), dtype=torch.float)
    else:
        conf = mol.GetConformer()
        pos = torch.tensor(conf.GetPositions(), dtype=torch.float)

    # Node features: Atomic Number, Hybridization, Formal Charge, Aromaticity
    atom_features = []
    for atom in mol.GetAtoms():
        atom_features.append([
            float(atom.GetAtomicNum()),
            float(atom.GetDegree()),
            float(atom.GetFormalCharge()),
            float(atom.GetIsAromatic())
        ])
    x = torch.tensor(atom_features, dtype=torch.float)

    # Chemical Bonds (Edges)
    edge_list = []
    for bond in mol.GetBonds():
        i, j = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
        edge_list.extend([[i, j], [j, i]])
    
    if len(edge_list) == 0:
        edge_index = torch.empty((2, 0), dtype=torch.long)
    else:
        edge_index = torch.tensor(edge_list, dtype=torch.long).t().contiguous()

    return Data(x=x, pos=pos, edge_index=edge_index, smiles=smiles)

graph_corpus = [mol_to_spatial_graph(s) for s in df['smiles'] if mol_to_spatial_graph(s) is not None]
```

### Phase 3: Lightweight SE(3)-Equivariant Encoder & Decoder

Implement the $\mathrm{SE}(3)$ message passing layers directly via Equivariant Graph Convolutional Layers (EGNN), satisfying your patent's coordinate transformation claim:

$$\mathbf{x}_i^{(l+1)} = \mathbf{x}_i^{(l)} + C \sum_{j \in \mathcal{N}(i)} (\mathbf{x}_i^{(l)} - \mathbf{x}_j^{(l)}) \cdot \phi_x(\mathbf{m}_{ij}^{(l)})$$

```python
class EGNNLayer(nn.Module):
    """SE(3)-Equivariant Message Passing Layer (Module 102)."""
    def __init__(self, feat_dim):
        super().__init__()
        self.phi_m = nn.Sequential(nn.Linear(feat_dim * 2 + 1, feat_dim), nn.SiLU())
        self.phi_x = nn.Sequential(nn.Linear(feat_dim, 1), nn.Tanh())
        self.phi_h = nn.Sequential(nn.Linear(feat_dim * 2, feat_dim), nn.SiLU())

    def forward(self, h, pos, edge_index):
        row, col = edge_index
        dist_sq = torch.sum((pos[row] - pos[col]) ** 2, dim=-1, keepdim=True)
        m_ij = self.phi_m(torch.cat([h[row], h[col], dist_sq], dim=-1))
        
        # Equivariant coordinate update: transforms with 3D rotation/translation
        coord_trans = (pos[row] - pos[col]) * self.phi_x(m_ij)
        pos_out = pos.clone()
        pos_out.index_add_(0, row, coord_trans)
        
        # Invariant scalar feature update
        m_agg = torch.zeros_like(h).index_add_(0, row, m_ij)
        h_out = self.phi_h(torch.cat([h, m_agg], dim=-1))
        return h_out, pos_out

class SE3MolecularVAE(nn.Module):
    def __init__(self, in_dim=4, hidden_dim=32, latent_dim=16):
        super().__init__()
        self.embedding = nn.Linear(in_dim, hidden_dim)
        self.layer1 = EGNNLayer(hidden_dim)
        self.layer2 = EGNNLayer(hidden_dim)
        
        # Mean and Log-variance Projection Heads (FIG. 3)
        self.mu_head = nn.Linear(hidden_dim, latent_dim)
        self.logvar_head = nn.Linear(hidden_dim, latent_dim)
        
        # Property Predictor Head (Module 108 / theta_P) with Monte Carlo Dropout
        self.surrogate_predictor = nn.Sequential(
            nn.Linear(latent_dim, 32),
            nn.Dropout(p=0.2),
            nn.ReLU(),
            nn.Linear(32, 1) # Surrogate Bioactivity Prioritization Score
        )

    def encode(self, x, pos, edge_index, batch):
        h = self.embedding(x)
        h, pos = self.layer1(h, pos, edge_index)
        h, pos = self.layer2(h, pos, edge_index)
        
        # Permutation-invariant Global Readout
        batch_size = int(batch.max().item()) + 1
        h_g = torch.zeros(batch_size, h.size(1), device=h.device).scatter_add_(
            0, batch.unsqueeze(1).expand_as(h), h
        )
        return self.mu_head(h_g), self.logvar_head(h_g)

    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def forward(self, batch_data):
        mu, logvar = self.encode(batch_data.x, batch_data.pos, batch_data.edge_index, batch_data.batch)
        z = self.reparameterize(mu, logvar)
        pred_bioactivity = self.surrogate_predictor(z)
        return z, mu, logvar, pred_bioactivity

```

### Phase 4: In-Silico Evaluator & UCB Active Learning Acquisition

Implement the scalarized composite utility engine and active learning selection, avoiding unverified binding claims while retaining programmatic rigor:

```python
def compute_synthetic_accessibility(mol):
    # Approximation of fragment-penalty SA score between 1 (easy) and 10 (hard)
    num_rings = mol.GetRingInfo().NumRings()
    num_chiral = len(Chem.FindMolChiralCenters(mol, includeUnassigned=True))
    sa = 1.0 + (num_rings * 0.5) + (num_chiral * 0.8) + (mol.GetNumAtoms() * 0.05)
    return min(max(sa, 1.0), 10.0)

def evaluate_candidate_batch(generated_smiles, model, num_mc_passes=10):
    evaluated_records = []
    model.eval()
    
    # Enable dropout during inference to compute Epistemic Uncertainty sigma_u
    for m in model.modules():
        if isinstance(m, nn.Dropout):
            m.train()
            
    for smi in generated_smiles:
        mol = Chem.MolFromSmiles(smi)
        if mol is None:
            continue
        g = mol_to_spatial_graph(smi)
        g.batch = torch.zeros(g.x.size(0), dtype=torch.long)
        
        # In-silico QED & SA scores
        qed_val = QED.qed(mol)
        sa_val = compute_synthetic_accessibility(mol)
        norm_sa = 1.0 - (sa_val / 10.0) # Higher is better
        
        # Monte Carlo Uncertainty Estimation
        preds = []
        with torch.no_grad():
            for _ in range(num_mc_passes):
                z, _, _, p_bio = model(g)
                preds.append(p_bio.item())
        
        mu_bio = float(pd.Series(preds).mean())
        sigma_bio = float(pd.Series(preds).std())
        
        # Scalarized Multi-Objective Utility Engine (FIG. 4):
        # u(z) = 0.4*QED + 0.3*SA_norm + 0.3*Predicted_Bioactivity
        utility = (0.4 * qed_val) + (0.3 * norm_sa) + (0.3 * mu_bio)
        
        # Upper Confidence Bound (UCB) Policy: alpha_UCB = mu + kappa * sigma
        kappa = 0.7
        ucb_score = utility + kappa * sigma_bio
        
        evaluated_records.append({
            'graph': g,
            'smiles': smi,
            'qed': qed_val,
            'sa': sa_val,
            'predicted_bio': mu_bio,
            'uncertainty': sigma_bio,
            'ucb_score': ucb_score
        })
        
    return evaluated_records
```

### Phase 5: Empirical Fisher Matrix Calculation & Closed-Loop EWC Regularization

Compute the diagonal Fisher matrix $F_j$ over baseline DrugBank data ($\mathcal{D}_{\text{chem}}$) and apply the quadratic preservation penalty:

$$\mathcal{L}_{\text{total}}(\Theta) = \mathcal{L}_{\text{task}} + \beta \mathcal{L}_{\text{NELBO}} + \sum_{j=1}^P \frac{\lambda_{\text{EWC}}}{2} F_j (\Theta_j - \Theta_j^*)^2$$

```python
def compute_diagonal_fisher(model, dataloader):
    """Computes empirical Fisher coefficients F_j over D_chem (FIG. 5)."""
    model.eval()
    fisher = {n: torch.zeros_like(p) for n, p in model.named_parameters() if p.requires_grad}
    
    for batch in dataloader:
        model.zero_grad()
        _, mu, logvar, _ = model(batch)
        # Negative ELBO reconstruction proxy
        kl_loss = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
        kl_loss.backward()
        
        for n, p in model.named_parameters():
            if p.grad is not None:
                fisher[n] += p.grad.data.pow(2) / len(dataloader)
    return fisher

def regularized_adaptation_step(model, active_batch, fisher, reference_params, optimizer, lambda_ewc=500.0):
    model.train()
    optimizer.zero_grad()
    
    # 1. Dynamic Re-Encoding of active batch (FIG. 5)
    z, mu, logvar, pred_bio = model(active_batch)
    
    # Task Loss: optimize against top-prioritized active candidates
    target_rewards = torch.tensor([item['ucb_score'] for item in active_batch.records], dtype=torch.float).unsqueeze(1)
    loss_task = nn.MSELoss()(pred_bio, target_rewards)
    
    # Variational prior maintenance
    loss_nelbo = -0.5 * torch.mean(1 + logvar - mu.pow(2) - logvar.exp())
    
    # 2. Parameter-Preservation Regularization (EWC Penalty)
    loss_ewc = 0.0
    for n, p in model.named_parameters():
        if n in fisher:
            loss_ewc += (fisher[n] * (p - reference_params[n]).pow(2)).sum()
            
    # Total Composite Adaptation Loss
    loss_total = loss_task + 0.1 * loss_nelbo + (lambda_ewc / 2.0) * loss_ewc
    loss_total.backward()
    optimizer.step()
    
    return loss_total.item(), loss_task.item(), loss_ewc.item()

```
